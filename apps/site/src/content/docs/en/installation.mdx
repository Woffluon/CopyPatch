export const npmInstall = `npm install @copypatch/core @copypatch/react @copypatch/backend @copypatch/storage-sqlite @copypatch/next`;

export const cliUsage = `# Interactive scaffolding (or provide explicit flags)
npx copypatch init --framework next --storage sqlite

# Test scaffolding without writing files
npx copypatch init --framework next --storage sqlite --dry-run

# Generate secure Argon2id passphrase hash
printf 'your-secure-passphrase' | npx copypatch hash --stdin

# Apply database schema migrations
npx copypatch migrate --storage sqlite

# Verify runtime configuration and health
npx copypatch doctor`;

<h1>Install CopyPatch v2</h1>
  <p class="lead">
    CopyPatch packages are published under the <code>@copypatch</code> npm scope. Install only the packages required for your client and server architecture.
  </p>

  <h2>Prerequisites</h2>
  <ul>
    <li><strong>Runtime:</strong> Node.js <code>&gt;=20.0.0</code> (Node.js 24 LTS recommended)</li>
    <li><strong>Module Format:</strong> Pure ESM (ECMAScript Modules)</li>
    <li><strong>Peer Dependencies:</strong> React <code>&gt;=18.0.0</code> and React DOM <code>&gt;=18.0.0</code></li>
    <li><strong>Environment:</strong> Server-capable host application with same-origin routing</li>
  </ul>

  <h2>CLI Command Reference</h2>
  <table>
    <thead>
      <tr>
        <th>Command</th>
        <th>Flags</th>
        <th>Purpose</th>
      </tr>
    </thead>
    <tbody>
      <tr>
        <td><code>copypatch init</code></td>
        <td><code>--framework &lt;next|astro|react-router|vite-node&gt;</code><br/><code>--storage &lt;sqlite|postgres&gt;</code><br/><code>--dry-run</code>, <code>--cwd &lt;dir&gt;</code></td>
        <td>Generates <code>copypatch.config.ts</code>, <code>copypatch.env.example</code>, and the matching route file. Safe to re-run; never overwrites conflicting files.</td>
      </tr>
      <tr>
        <td><code>copypatch hash</code></td>
        <td><code>--stdin</code>, <code>--passphrase &lt;str&gt;</code></td>
        <td>Generates a cryptographically secure Argon2id hash. Using <code>--stdin</code> prevents exposing secrets in shell history.</td>
      </tr>
      <tr>
        <td><code>copypatch migrate</code></td>
        <td><code>--storage &lt;sqlite|postgres&gt;</code></td>
        <td>Executes idempotent database schema migrations.</td>
      </tr>
      <tr>
        <td><code>copypatch doctor</code></td>
        <td>None</td>
        <td>Inspects the local configuration, environment variables, and storage connectivity.</td>
      </tr>
    </tbody>
  </table>

  <blockquote>
    <strong>Note:</strong> The CopyPatch CLI intentionally does not have a <code>serve</code> command. CopyPatch is embedded into your application's existing HTTP server instead of running as a separate daemon.
  </blockquote>
