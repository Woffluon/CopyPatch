import CodeBlock from '../../../components/ui/CodeBlock.astro';

export const nextLocaleExample = `// app/[locale]/layout.tsx
import { NextCopyPatchProvider } from '@copypatch/next';
import { readPublishedSnapshot } from '@copypatch/next/server';
import { copypatch } from '@/lib/copypatch';

export default async function LocaleLayout({
  children,
  params,
}: {
  children: React.ReactNode;
  params: Promise<{ locale: string }>;
}) {
  const { locale } = await params;
  const snapshot = await readPublishedSnapshot(copypatch, locale);

  return (
    <html lang={locale}>
      <body>
        <NextCopyPatchProvider locale={locale} initialSnapshot={snapshot}>
          {children}
        </NextCopyPatchProvider>
      </body>
    </html>
  );
}`;

export const componentExample = `import { EditableText, useCopyPatch } from '@copypatch/react';

export function PricingHeader({ locale }: { locale: string }) {
  // English fallback: "Simple Pricing" / Turkish fallback: "Basit Fiyatlandırma"
  const defaultTitle = locale === 'tr' ? 'Basit Fiyatlandırma' : 'Simple Pricing';
  const ctaLabel = useCopyPatch('pricing.cta', locale === 'tr' ? 'Hemen Başla' : 'Start Now');

  return (
    <div className="pricing-header">
      <EditableText contentKey="pricing.title" as="h2">
        {defaultTitle}
      </EditableText>
      <button type="button">{ctaLabel}</button>
    </div>
  );
}`;

<h1>Multi-Locale & Translation Isolation</h1>
  <p class="lead">
    CopyPatch provides strict per-locale data isolation. Each language code (e.g. <code>en</code>, <code>tr</code>, <code>de</code>, <code>es</code>) maintains its own independent published snapshot, draft revision, and edit history.
  </p>

  <h2>How Locale Isolation Works</h2>
  <ul>
    <li><strong>Zero Cross-Contamination:</strong> Publishing edits in <code>tr</code> has no effect on <code>en</code>, <code>de</code>, or any other locale.</li>
    <li><strong>Independent Revision Counters:</strong> Revision numbers are tracked per locale. If two editors work on English and Turkish concurrently, their revisions never conflict.</li>
    <li><strong>Fallback Resolution:</strong> When a key has not yet been edited in storage for a given locale, CopyPatch gracefully renders the component's default fallback text.</li>
  </ul>

  <h2>Locale Format & Validation Rules</h2>
  <p>
    CopyPatch validates locale strings using standard BCP 47 and alphanumeric tag conventions before performing any storage read or mutation:
  </p>
  <table>
    <thead>
      <tr>
        <th>Valid Examples</th>
        <th>Description</th>
      </tr>
    </thead>
    <tbody>
      <tr>
        <td><code>en</code>, <code>tr</code>, <code>de</code>, <code>ja</code></td>
        <td>Two-letter ISO 639-1 language codes.</td>
      </tr>
      <tr>
        <td><code>en-US</code>, <code>pt-BR</code>, <code>zh-CN</code></td>
        <td>Language and regional country subtags.</td>
      </tr>
    </tbody>
  </table>

  <p>
    Underscore-separated POSIX forms such as <code>en_US</code> and <code>tr_TR</code> are rejected. Convert them to hyphenated BCP 47 tags before passing them to CopyPatch.
  </p>
