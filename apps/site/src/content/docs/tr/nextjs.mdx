export const backendSingleton = `// lib/copypatch.ts
import { createCopyPatchBackend } from '@copypatch/backend';
import { createSQLitePersistence } from '@copypatch/storage-sqlite';

const persistence = createSQLitePersistence('./copypatch.sqlite');
await persistence.migrate();

const copypatch = createCopyPatchBackend({
  persistence,
  passphraseHash: process.env.COPYPATCH_PASSPHRASE_HASH!,
});`;

export const routeHandler = `// app/%5F%5Fcopypatch/api/v2/[...path]/route.ts
import { createCopyPatchRouteHandlers } from '@copypatch/next/server';
import { copypatch } from '@/lib/copypatch';

// Bunu deployment platformunuzun sağladığı ya da forwarding header'larına
// açıkça güvendiğiniz bir proxy'nin doğrulanmış adresiyle uygulayın.
declare function resolveTrustedClientAddress(request: Request): Promise<string>;

export const { GET, POST, PUT, PATCH, DELETE, HEAD, OPTIONS } =
  createCopyPatchRouteHandlers(copypatch, {
    resolveContext: async (request) => ({
      clientAddress: await resolveTrustedClientAddress(request),
    }),
  });`;

export const serverPage = `// app/[locale]/page.tsx
import { NextCopyPatchProvider, EditableText } from '@copypatch/next';
import { readPublishedSnapshot } from '@copypatch/next/server';
import { copypatch } from '@/lib/copypatch';

export default async function Page({
  params,
}: {
  params: Promise<{ locale: string }>;
}) {
  const { locale } = await params;

  // Sunucu Bileşeni içinde doğrudan veritabanı okuması (Sıfır HTTP gecikmesi)
  const initialSnapshot = await readPublishedSnapshot(copypatch, locale, {
    fallback: {
      revision: 1,
      content: { 'home.hero.title': 'Platformumuza Hoş Geldiniz' },
    },
  });

  return (
    <NextCopyPatchProvider locale={locale} initialSnapshot={initialSnapshot}>
      <main className="container">
        <EditableText contentKey="home.hero.title" as="h1">
          Platformumuza Hoş Geldiniz
        </EditableText>
      </main>
    </NextCopyPatchProvider>
  );
}`;

  <CodeBlock code={routeHandler} lang="ts" title="app/%5F%5Fcopypatch/api/v2/[...path]/route.ts" />

  <p>
    Güvenilir bir <code>clientAddress</code> çözümlenmezse unsafe istekler <code>CLIENT_ADDRESS_UNAVAILABLE</code> ile fail-closed olur. CopyPatch <code>x-forwarded-for</code> başlığına otomatik olarak güvenmez. Deployment'ınız bilinçli olarak tek bir paylaşımlı rate-limit bucket'ı kabul ediyorsa bunun yerine <code>unsafeRequestWithoutClientAddress: 'shared-bucket'</code> seçeneğini açıkça kullanın.
  </p>

  <blockquote>
    <strong>Neden <code>%5F%5Fcopypatch</code>?</strong> Next.js dosya sisteminde alt çizgi ile başlayan klasörleri (örn. <code>_copypatch</code>) özel (private) kabul eder ve bunları URL yönlendirmesinden hariç tutar. Alt çizgileri yüzde kodlamasıyla (URL percent-encoding) <code>%5F%5Fcopypatch</code> olarak adlandırarak, Next.js'in standart genel <code>/__copypatch/api/v2/...</code> yolunu sorunsuz sunması sağlanır.
  </blockquote>

  <h2>Next.js Entegrasyonunun Avantajları</h2>
  <ul>
    <li><strong>Sıfır Yerleşim Kayması (Zero Layout Shift):</strong> Metinler sunucu tarafında oluşturulan ilk HTML yanıtına gömülür.</li>
    <li><strong>Sıfır HTTP Ek Yükü:</strong> Sunucu bileşeni <code>localhost</code> adresine istek göndermeden doğrudan bellek veya veritabanı havuzundan okur.</li>
    <li><strong>Proxy veya Rewrite İhtiyacı Yok:</strong> Rota doğrudan Next.js uygulamanızın içinde yaşadığı için <code>next.config.js</code> içinde karmaşık yönlendirme kuralları gerekmez.</li>
  </ul>

  <h2>Statik Dışa Aktarma (Static Export) Uyarısı</h2>
  <blockquote>
    <strong>Önemli:</strong> CopyPatch, serverless Node deployment'ları dahil Node.js sunucu runtime'ı gerektirir. Edge ve <code>output: 'export'</code> çalışma zamanları kimlik doğrulamalı düzenleme API'sini veya native veritabanı kalıcılığını barındıramaz.
  </blockquote>
