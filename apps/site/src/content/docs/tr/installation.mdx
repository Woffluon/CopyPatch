export const pnpmInstall = `# Çekirdek ve React İstemcisi
pnpm add @copypatch/core @copypatch/react

# Gömülü Backend ve SQLite Kalıcılık Katmanı
pnpm add @copypatch/backend @copypatch/storage-sqlite

# Çerçeve Adaptörleri (Uygulamanıza uygun olanı seçin)
pnpm add @copypatch/next    # Next.js App Router için
pnpm add @copypatch/node    # Astro, Express, Fastify, Hono, React Router için`;

export const npmInstall = `npm install @copypatch/core @copypatch/react @copypatch/backend @copypatch/storage-sqlite @copypatch/node @copypatch/next`;

export const cliUsage = `# İnteraktif şablon oluşturma (veya doğrudan bayraklarla çalıştırma)
npx copypatch init --framework next --storage sqlite

# Dosyalara yazmadan şablonu test etme (Kuru çalıştırma)
npx copypatch init --framework next --storage sqlite --dry-run

# Güvenli Argon2id parola özeti (hash) üretme
printf 'guclu-ve-guvenli-parolaniz' | npx copypatch hash --stdin

# Veritabanı şema migrasyonlarını çalıştırma
npx copypatch migrate --storage sqlite

# Çalışma zamanı yapılandırmasını ve sağlık durumunu denetleme
npx copypatch doctor`;

<h1>CopyPatch v3 Kurulumu</h1>
  <p class="lead">
    CopyPatch paketleri <code>@copypatch</code> npm kapsamı altında yayınlanmaktadır. Uygulamanızın istemci ve sunucu mimarisine göre yalnızca ihtiyaç duyduğunuz paketleri projenize ekleyin.
  </p>

  <h2>Önkoşullar</h2>
  <ul>
    <li><strong>Çalışma Zamanı (Runtime):</strong> Node.js <code>&gt;=20.0.0</code> (Node.js 24 LTS önerilir)</li>
    <li><strong>Modül Formatı:</strong> Saf ESM (ECMAScript Modules)</li>
    <li><strong>Eş Bağımlılıklar (Peer Dependencies):</strong> React ve React DOM <code>&gt;=18 &lt;20</code></li>
    <li><strong>Çalışma Ortamı:</strong> Aynı kökenden (same-origin) API rotası barındırabilen sunucu özellikli bir web uygulaması</li>
  </ul>

  <h2>CLI Komut Başvuru Tablosu</h2>
  <table>
    <thead>
      <tr>
        <th>Komut</th>
        <th>Bayraklar (Flags)</th>

    </thead>
    <tbody>
      <tr>
        <td><code>copypatch init</code></td>
        <td><code>--framework &lt;next|astro|react-router|vite-node&gt;</code><br/><code>--storage &lt;sqlite|postgres&gt;</code><br/><code>--dry-run</code>, <code>--cwd &lt;dizin&gt;</code></td>

        <td><code>--stdin</code>, <code>--passphrase &lt;metin&gt;</code></td>

      <tr>
        <td><code>copypatch migrate</code></td>
        <td><code>--storage &lt;sqlite|postgres&gt;</code></td>
        <td>Seçilen veritabanı motoru için sürüm takipli ve idempotent şema migrasyonlarını çalıştırır.</td>
      </tr>
      <tr>
        <td><code>copypatch doctor</code></td>
        <td>Yok</td>

      </tr>
    </tbody>
  </table>

  <blockquote>
    <strong>Önemli Not:</strong> CopyPatch CLI aracında kasıtlı olarak bir <code>serve</code> komutu bulunmaz. CopyPatch bağımsız bir arka plan süreci olarak değil, doğrudan mevcut web uygulamanızın HTTP sunucusunun içine gömülü olarak çalışır.
  </blockquote>
