export const basicBackend = `import { createCopyPatchBackend } from '@copypatch/backend';
import { createSQLitePersistence } from '@copypatch/storage-sqlite';

const persistence = createSQLitePersistence('./copypatch.sqlite');
await persistence.migrate();

const copypatch = createCopyPatchBackend({
  persistence,
  passphraseHash: process.env.COPYPATCH_PASSPHRASE_HASH!,
});`;

export const customAuthAdapter = `import { createCopyPatchBackend } from '@copypatch/backend';
import { type CopyPatchAuthAdapter } from '@copypatch/core';
import { createPostgresPersistence } from '@copypatch/storage-postgres';

// Mevcut kimlik doğrulama sisteminizi (NextAuth, Clerk, Auth0 vb.) bağlayan özel adaptör
interface HostAuthContext {
  user?: {
    id: string;
    isAdmin: boolean;
    canEditCopy: boolean;
    canPublishCopy: boolean;
  };
  csrfValid?: boolean;
}

const hostAuthAdapter: CopyPatchAuthAdapter<HostAuthContext> = {
  async authenticate(_request, context) {
    const user = context.hostAuth?.user;
    if (!user) return null;

    const roles: ('editor' | 'publisher')[] = [];
    if (user.isAdmin || user.canEditCopy) roles.push('editor');
    if (user.isAdmin || user.canPublishCopy) roles.push('publisher');

    return { subject: user.id, roles };
  },

  async verifyMutation(_request, _principal, context) {
    // Ana uygulamanın CSRF belirtecini veya oturum geçerliliğini denetleyin
    return context.hostAuth?.csrfValid === true;
  },
};

const backend = createCopyPatchBackend({
  persistence: createPostgresPersistence({ connectionString: process.env.COPYPATCH_DATABASE_URL! }),
  authAdapter: hostAuthAdapter,
});`;

  <h2>2. Kimlik Doğrulama Stratejileri</h2>
  <p>
    <code>createCopyPatchBackend</code> birbirinden bağımsız iki stratejiden tam olarak birini kabul eder:
  </p>

  <h3>Strateji A: Dahili Parola Oturumu (<code>passphraseHash</code>)</h3>
  <ul>
    <li>Mevcut kullanıcı veritabanı olmayan projeler için CopyPatch tarafından tamamen bağımsız yönetilir.</li>
    <li><strong>Argon2id</strong> parola özeti (RFC 9106 önerilen parametreleri) kullanır.</li>
    <li><code>HttpOnly</code>, <code>SameSite=Strict</code> oturum çerezleri ve çift belirteçli (dual-token) CSRF koruması üretir.</li>
    <li>Müşteri teslimatları, ajans projeleri ve pazarlama web siteleri için idealdir.</li>
  </ul>

  <h3>Strateji B: Host Kimlik Doğrulama Adaptörü (<code>authAdapter</code>)</h3>
  <ul>
    <li>Kullanıcı çözümlemesini ve rol atamasını mevcut kimlik sağlayıcınıza (NextAuth, Lucia, Clerk, Auth0, Supabase vb.) devreder.</li>
    <li>Kullanıcı yetkilerinizi CopyPatch'in <code>editor</code> ve <code>publisher</code> rollerine eşler.</li>
    <li>Veri değiştiren isteklerin güvenlik doğrulamasını ana uygulamanın mantığına bırakır.</li>
  </ul>

  <h2>4. Node Çerçeve Adaptörleri</h2>
  <p>
    <code>@copypatch/node</code> paketi, popüler Node.js sunucu çatıları için hafif HTTP köprüleri sağlar:
  </p>
  <ul>
    <li><code>createNodeHandler(backend)</code> – Yerel Node.js <code>http.createServer</code></li>
    <li><code>expressMiddleware(backend)</code> – Express ara yazılımı</li>
    <li><code>fastifyCopyPatchHandler(backend)</code> – Fastify rota işleyicisi</li>
    <li><code>createHonoHandler(backend)</code> – Hono Web standart işleyicisi</li>
  </ul>

    <strong>Geçmiş Sürüm Notu:</strong> <code>@copypatch/server</code> v1 sürümündeki bağımsız sunucu süreciydi. v2'de yerini tamamen gömülü çalışan <code>@copypatch/backend</code> ve <code>@copypatch/node</code> kütüphanelerine bırakmıştır. Yayınlanmış eski paketler npm üzerinde kalacaktır; yeni projelerde doğrudan v2 kullanılmalıdır.
