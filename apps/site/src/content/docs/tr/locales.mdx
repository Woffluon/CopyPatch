import CodeBlock from '../../../components/ui/CodeBlock.astro';

export const nextLocaleExample = `// app/[locale]/layout.tsx
import { NextCopyPatchProvider } from '@copypatch/next';
import { readPublishedSnapshot } from '@copypatch/next/server';
import { copypatch } from '@/lib/copypatch';

export default async function LocaleLayout({
  children,
  params,
}: {
  children: React.ReactNode;
  params: Promise<{ locale: string }>;
}) {
  const { locale } = await params;
  const snapshot = await readPublishedSnapshot(copypatch, locale);

  return (
    <html lang={locale}>
      <body>
        <NextCopyPatchProvider locale={locale} initialSnapshot={snapshot}>
          {children}
        </NextCopyPatchProvider>
      </body>
    </html>
  );
}`;

export const componentExample = `import { EditableText, useCopyPatch } from '@copypatch/react';

export function FiyatlandirmaBasligi({ locale }: { locale: string }) {
  // Türkçe varsayılan: "Şeffaf Fiyatlandırma" / İngilizce varsayılan: "Transparent Pricing"
  const defaultTitle = locale === 'tr' ? 'Şeffaf Fiyatlandırma' : 'Transparent Pricing';
  const ctaLabel = useCopyPatch('pricing.cta', locale === 'tr' ? 'Hemen Başla' : 'Start Now');

  return (
    <div className="pricing-header">
      <EditableText contentKey="pricing.title" as="h2">
        {defaultTitle}
      </EditableText>
      <button type="button">{ctaLabel}</button>
    </div>
  );
}`;

<h1>Çoklu Dil ve Çeviri İzolasyonu</h1>
  <p class="lead">
    CopyPatch, diller arasında tam veri izolasyonu sağlar. Her dil kodu (örn. <code>tr</code>, <code>en</code>, <code>de</code>, <code>es</code>) veritabanında kendi bağımsız yayın anlık görüntüsünü, taslak revizyonunu ve düzenleme geçmişini barındırır.
  </p>

  <h2>Dil İzolasyonu Nasıl Çalışır?</h2>
  <ul>
    <li><strong>Sıfır Çapraz Etkileşim:</strong> <code>tr</code> dilinde yapılan ve yayınlanan bir düzenleme, <code>en</code>, <code>de</code> veya başka herhangi bir dildeki metinleri asla etkilemez.</li>
    <li><strong>Bağımsız Revizyon Sayaçları:</strong> Revizyon sayaçları her dil için ayrı tutulur. Türkçe ve İngilizce dillerinde çalışan iki ayrı editör birbirlerinin çalışmalarında asla revizyon çakışması (conflict) yaşamaz.</li>
    <li><strong>Yedek (Fallback) Metin Desteği:</strong> Bir dil için veritabanında henüz yayınlanmış bir metin kaydı yoksa, CopyPatch bileşenin JSX içinde sağladığı varsayılan yedek metni güvenle render eder.</li>
  </ul>
