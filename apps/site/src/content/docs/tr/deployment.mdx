export const sqliteConfig = `import { createSQLitePersistence } from '@copypatch/storage-sqlite';

const persistence = createSQLitePersistence({
  filename: process.env.COPYPATCH_SQLITE_PATH ?? './data/copypatch.sqlite',
  busyTimeoutMs: 5_000,
});

// Sunucu başlarken şema migrasyonlarını çalıştırın
await persistence.migrate();`;

export const postgresConfig = `import { Pool } from 'pg';
import { createPostgresPersistence } from '@copypatch/storage-postgres';

const pool = new Pool({
  connectionString: process.env.DATABASE_URL!,
  max: 10,
  idleTimeoutMillis: 30000,
  ssl: process.env.NODE_ENV === 'production' ? { rejectUnauthorized: true } : false,
});

const persistence = createPostgresPersistence({
  pool,
  schema: 'copypatch',
});

// Idempotent şema migrasyonları (PostgreSQL advisory transaction kilitleri ile korunur)
await persistence.migrate();`;

<h1>Üretim Ortamı ve Dağıtım Rehberi</h1>
  <p class="lead">
    CopyPatch v2, doğrudan ana web uygulamanızın sunucusu içinde barındırılır. API rotası aynı HTTPS kökeni (same-origin) altında <code>/__copypatch/api/v2</code> adresinde çalıştığı için ek süreçler, açık portlar veya CORS reverse proxy kuralları gerektirmez.
  </p>

  <h2>1. Kalıcılık (Depolama) Yapılandırması</h2>

  <h3>Seçenek A: SQLite (Tek Node Sunucusu / VPS / Konteyner)</h3>
  <p>
    SQLite tek sunuculu yapılar için son derece hızlı, işlemsel ve sıfır bakım maliyetlidir:
  </p>
  <CodeBlock code={sqliteConfig} lang="ts" title="lib/copypatch.ts (SQLite)" />
  <ul>
    <li><strong>Kalıcı Disk (Persistent Volume):</strong> Docker veya Kubernetes üzerinde veritabanı dosyasını ve WAL yan dosyalarını (<code>.sqlite-wal</code> ve <code>.sqlite-shm</code>) korumak için kalıcı bir dizin (örn. <code>/data</code>) bağlayın.</li>
    <li><strong>WAL Modu:</strong> <code>@copypatch/storage-sqlite</code> yüksek eşzamanlılık için WAL (Write-Ahead Logging) modunu ve meşgul yeniden deneme sürelerini otomatik olarak yönetir.</li>
  </ul>

  <h3>Seçenek B: PostgreSQL (Çoklu Sunucu Kümeleri / Sunucusuz Mimari)</h3>
  <p>
    PostgreSQL, birden fazla sunucu örneğinde metinleri, taslakları ve hız sınırlarını senkronize eder:
  </p>
  <CodeBlock code={postgresConfig} lang="ts" title="lib/copypatch.ts (PostgreSQL)" />
  <ul>
    <li><strong>Advisory Kilitleri:</strong> Şema migrasyonları PostgreSQL transaction advisory kilitleri kullanarak eşzamanlı konteyner dağıtımlarında mükerrer migrasyon riskini engeller.</li>
    <li><strong>Bağlantı Havuzu:</strong> Mevcut bir <code>pg.Pool</code> nesnesini aktararak ana uygulamanızla veritabanı bağlantılarını paylaşabilirsiniz.</li>
  </ul>

  <h2>2. Ortam Değişkenleri Referansı</h2>

  <h2>4. Üretim Dağıtım Kontrol Listesi</h2>
  <ul>
    <li><strong>HTTPS Zorunluluğu:</strong> <code>Secure</code> ve <code>__Host-</code> oturum çerezlerinin çalışması için uygulamanızı HTTPS üzerinden sunun.</li>
    <li><strong>Açılışta Migrasyon:</strong> Trafik kabul etmeden önce sunucu başlatma kodunuzda <code>await persistence.migrate()</code> fonksiyonunu çağırın.</li>
    <li><strong>Statik SSG ile Düzenleme Yapılamaz:</strong> Barındırma ortamınızın API rotasını Node.js veya sunucusuz fonksiyon olarak yürüttüğünden emin olun.</li>
    <li><strong>Düzenli Yedekleme:</strong> SQLite disk hacminizi veya PostgreSQL veritabanınızı otomatik periyodik yedekleme planınıza dahil edin.</li>
  </ul>
